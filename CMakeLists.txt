foreach(ASSET_KIND IN ITEMS FONT ICON)
  string(TOLOWER "${ASSET_KIND}" ASSET_NAME)
  set(ASSET_MANIFEST "${CMAKE_CURRENT_BINARY_DIR}/generated/${ASSET_NAME}_inputs.cmake")
  # Track additions/removals as well as edits, without touching an unchanged list.
  file(CONFIGURE OUTPUT "${ASSET_MANIFEST}"
    CONTENT "set(CABINET_ASSET_FILES [==[${CABINET_${ASSET_KIND}_FILES}]==])\n" @ONLY)
  add_custom_command(OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/generated/${ASSET_NAME}_data.cpp"
    COMMAND "${CMAKE_COMMAND}" "-DASSET_KIND=${ASSET_KIND}"
      "-DASSET_MANIFEST=${ASSET_MANIFEST}"
      "-DASSET_OUTPUT=${CMAKE_CURRENT_BINARY_DIR}/generated/${ASSET_NAME}_data.cpp"
      -P "${CMAKE_CURRENT_SOURCE_DIR}/cmake/EmbedAssets.cmake"
    DEPENDS ${CABINET_${ASSET_KIND}_FILES} "${ASSET_MANIFEST}"
      "${CMAKE_CURRENT_SOURCE_DIR}/cmake/EmbedAssets.cmake"
    COMMENT "Embedding ${ASSET_NAME} assets" VERBATIM)
endforeach()
